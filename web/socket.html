Định dạng message

        
// hello: gửi 1 lần khi kết nối
{"type":"hello","client_id":3,"version":"1.0.0","topics":["status","events"],
 "available_topics":["status","people","events","tracks"],"cameras":[{"id":"camera_1","name":"Camera 1","enabled":true}]}

// status: định kỳ (status_hz)
{"type":"status","ts":1791877285.9,
 "totals":{"persons":3,"inside":2,"outside":1,"entries":42,"exits":38,"objects":7,"alerts":0},
 "cameras":[{"id":"camera_1","name":"Camera 1","status":"running","fps":14.9,
             "persons":3,"inside":2,"outside":1,"mode":"zone","entries":42,"exits":38,
             "alerts":0,"objects":{"backpack":1,"glasses":2}}]}
// persons = đang thấy; inside / outside = số người đang ở trong / ngoài vùng lúc này
//   (mode "line": phía có điểm I là trong; null khi camera chưa vẽ line/vùng)
// persons - inside - outside = người mới xuất hiện, chưa xác định phía
// entries / exits = tổng lượt vào / ra từ lần reset gần nhất

// people: định kỳ (people_hz), toạ độ chuẩn hoá 0..1 theo frame_size
{"type":"people","ts":...,"cameras":[{"id":"camera_1","frame_size":[1920,1080],
 "people":[{"track_id":27,"side":"in","alert":false,"items":["backpack","glasses"],
            "objects":[{"label":"backpack","vi":"ba lô","category":"bags","conf":0.82,"alert":false,
                        "box":[0.43,0.38,0.51,0.60]},
                       {"label":"glasses","vi":"kính","category":"face","conf":0.71,"alert":false,
                        "box":[0.45,0.31,0.48,0.33]}],
            "conf":0.88,"box":[0.41,0.30,0.52,0.86],"foot":[0.47,0.85]}]}]}
// side: "in" | "out" | "unknown"; objects = đồ vật đang hiển thị (đã qua ngưỡng hiển thị + làm mượt)
// label gộp (vd hat, glasses, gun); không gửi trường clip

// events: gửi ngay khi xảy ra
{"type":"crossing","ts":...,"camera_id":"camera_1","track_id":27,"direction":"in","mode":"zone"}
{"type":"crossing","ts":...,"camera_id":"camera_1","track_id":27,"direction":"out","mode":"zone","dwell_s":42.5}
{"type":"alert","ts":...,"camera_id":"camera_1","track_id":31,"items":["knife"],"all_items":["backpack","knife"]}
{"type":"camera","ts":...,"camera_id":"camera_2","status":"offline","error":"stream stalled"}
{"type":"reset","ts":...,"camera_id":"camera_1"}

// tracks: tóm tắt 1 người sau khi rời khung (items = số khung thấy vật đó)
{"type":"track","ts":...,"camera_id":"camera_1","track_id":27,"first_ts":...,"last_ts":...,
 "duration":12.4,"frames":180,"items":{"backpack":120,"glasses":64},"alert":false,
 "first_side":"out","last_side":"in"}
